# Settings for "Play Astro Bot VR.bat", one key=value a line. Lines starting with # are ignored;
# what is commented out below shows the value that applies when the line is left out.

# The window at the start (menu=0 to go without it) sets resolution, framerate and fov below.

# The width of each eye's picture: 1440 is the console's largest (what a PlayStation 4 Pro
# draws, 1440x1536); 1800, 2160, 2520, 2880, 3240, 3600, 3960 or 4320 have the game draw that much larger
# (2880 = 2880x3072, four times the pixels), with the memory that takes. The game draws a step
# smaller by itself while the graphics card cannot keep up; dynamic=0 holds it to the size.
# game: the console's sizes, chosen by the game itself (816 to 1440).
#resolution=2880
#dynamic=1

# Full-refresh pacing is the default. Set headset refresh in SteamVR or Virtual Desktop.
# fps=full targets headset refresh. Numeric FPS choices set a maximum using whole refresh intervals.
#fps=full

# How much of the headset's field of view the game draws, in percent, 70 to 100: at 100 the
# picture fills all that the headset shows; less puts the same pixels over fewer degrees, a
# sharper picture with a dark border where the rest was.
#fov=100

# What fov is a percent of: headset (the headset being worn, whatever it is: the emulator asks
# it as the game starts) or psvr (a PlayStation VR's 100 by 103 degrees an eye, the view the
# game was made for, which leaves a narrow dark margin far out to the sides on a Quest 3).
#fov_of=headset

# 1: the window with the main settings comes up at every start.
#menu=1

# Sharpening of the picture on its way to the headset, 0 (none) to 1. Virtual Desktop has a
# sharpening of its own (Streaming tab): what is here comes on top of it.
#sharpen=0.3

# The most samples a pixel gets (1, 2, 4, 8). Left out: as the console draws it (4). With 1 the
# emulator smooths edges itself instead, unless antialias=0.
#msaa=4
#antialias=1

# 0: your hands are not used to place the controller in the game; it then stays at a fixed
# spot in front of you, turned by its own motion sensors. (The hands are what the headset
# tracks: hand tracking has to be on in the headset, and Virtual Desktop has to pass it on to
# the PC - its setting for forwarding tracking data. Where it does not, the places it gives
# the headset's controllers stand in, which it makes up from the hands while they lie unused.)
# The DualSense must be connected to this PC itself (USB cable or Bluetooth): paired with the
# headset, it reaches the PC through Virtual Desktop without motion sensors and touchpad.
#hands=1

# How far beyond the next picture the head position given to the game is predicted, in
# milliseconds (0 to 80). More if black edges show when you turn your head, less if the world
# wobbles.
#predict_ms=20

# 0: the right stick no longer doubles as a finger on the touchpad.
#stick_touchpad=1

# Makes what the microphone hears that many times louder for the game (0.1 to 30), for when
# blowing does too little. The launcher's window shows how loud the game hears things, and
# what it takes for blowing. (Holding the PS button and square blows without a microphone.)
#mic_gain=1

# 0: the headset's own controllers never stand in for a gamepad. (Otherwise they play while no
# gamepad is connected to the PC, and with one whenever they were used after it.)
#controllers=1

# Which of the headset's controllers is the controller in the game while they stand in for a
# gamepad: right or left.
#controller_hand=right

# The language the game is played in, as Windows names languages (fr-FR, de-DE, pt-BR, ja-JP
# ...). windows: the one Windows is shown in. The game has 28 languages and is in English
# for any other. The launcher's window has the list.
#language=windows

# How many degrees the view turns when L1 (the headset's controllers: the left grip) is held
# and the right stick flicked to a side, for those who cannot turn round where they sit.
# 0: never.
#turn=30

# 0: the game's surround sound is folded down to left and right instead of being rendered for
# the headset's speakers.
#surround=1

# 0: the game is never made to wait. (It waits, black, while the headset is off the head or
# shows something else, Virtual Desktop's view of the desktop for one, and goes on after.) For
# when it waits although the headset is on and shows the game.
#pause=1

# 0: the game counts time in frames, as on the console (slow motion whenever a frame takes
# longer than a sixtieth of a second).
#real_time=1

# The older way to say fps: how many of the headset's pictures every frame is given (1, 2, 3...).
# Takes the place of fps when set.
#pace=2

# Seconds to wait for a headset before the game starts on the monitor without one. A headset
# connected later is picked up anyway, wherever the game is by then. 0: do not wait.
#wait=60

# 0: do not look for a headset at all, play on the monitor.
#headset=1

# Where the game is, if it is not in the games folder next to this one: its eboot.bin, its
# folder, or its .pkg package (which is then unpacked into the games folder). The launcher
# writes this line itself when it has to ask where the game is.
#game=C:\path\to\CUSA12392\eboot.bin

# Extra environment variables for the emulator, as many lines as needed.
#env=NAME=value
# Precise guest timers default on; set 0 to compare the original timer path.
#env=SHADPS4_PERF_PRECISE_TIMERS=0
# Local headset desktop mirror defaults to 60 FPS; 0 disables, 1000 removes its cap.
#env=SHADPS4_VR_WINDOW_FPS=60
resolution=3960
fps=full
fov=100
menu=1
# Fix for effects showing in one eye only (upstream issue #18, confirmed 2026-11-01): the game is told a symmetric field of view (out=in, up=down) that covers all of the Frame's; its effect culling breaks on the Frame's lopsided one. Costs some sharpness: about a quarter of what is drawn falls outside the view.
env=SHADPS4_VR_FOV_SYMMETRIC=1
# Models keep their full detail twice as far as on the console (its distances suit its 1440-wide picture; capped at each level's draw cutoff); needs the 2026-11-01 build.
env=SHADPS4_TITLE_LOD=2
# The sun's shadow maps at 2048 (the game's own largest) in the levels that ask for 1024 or 1536: the bird, shark and spider bosses, the whale, the world select.
env=SHADPS4_TITLE_SHADOW_MAP_MIN=2048
# Hold the picture at the full resolution above (3960x4224 an eye). Without this the emulator drops to 3304 or 2640 wide whenever the game itself runs a little late at 120 Hz, though the GPU had room to spare (20-30% busy). Delete if the frame rate suffers.
dynamic=0
# Spot-light shadow maps at 1024 (the most the game has room for): the world select and one boss ask for 512, and levels entered from the world select kept 512. Needs the 2026-11-01 21:27 build.
env=SHADPS4_TITLE_SPOT_SHADOW_MAP_MIN=1024
# For shadow captures: when dump_render_targets is made, also save the sun shadow maps (50db00000), spot-light shadow maps (51db00000 with 8192 sun maps, 511b00000 with 4096, 50eb00000 with 2048) and Astro's drop-shadow masks (51fb00000, 513b00000 or 510b00000; 5a99c8000 or 5ca4d8000) raw. Costs nothing otherwise.
env=SHADPS4_DUMP_IMAGE=50db00000,51db00000,51fb00000,511b00000,513b00000,50eb00000,510b00000,5a99c8000,5ca4d8000
# The sun's near shadow cascade ends at 6 m at most (it reached 8 m in the canyons, 16-20 m in the storm and mine-cart levels): sharper shadows near you, the far cascade starting nearer. Delete to get the game's own split.
env=SHADPS4_TITLE_SHADOW_NEAR_SPLIT=6
# Sun shadow maps 8192x8192 where the level asks for 2048 (every level, with SHADOW_MAP_MIN=2048 above): 4x the shadow pixels a side of the game's own, for less crawling on shadow edges. 4096 also works (tested 2026-11-01: much better, some crawling left). Game code patch, needs the 2026-11-01 22:27 build. If the frame rate drops, shadows break or the game crashes, change 8192 to 4096, or delete the line for the game's 2048.
env=SHADPS4_TITLE_SUN_SHADOW_MAP=8192
# Frame microphone: blowing arrives at about -21 to -37 dB (mic test 2026-11-03); the game wants -21 dB for half strength, -9 dB for full. x5 was too weak (16% of a blow at full); x20 = +26 dB puts ~80% of a blow at full, the rest at half or more. Talking will blow too.
env=SHADPS4_MIC_GAIN=20
